## Review Focus

1. **Odd `Accept-Language`** (`en;q=0`, `*`, empty, `q=abc`) → falls back to `vi`, never throws. (Task 2 tests)
2. **Tampered cookie** (`locale=xx`, `locale=__proto__`) → ignored; `setLocale` with a bad value is a no-op. (Task 2 tests; Task 4 action guard)
3. **Zod built-in issue** (amount `Infinity`) → localized `invalid` text in both languages, never English zod text or a raw key. (Task 3 tests)
4. **User expense literally titled "Thanh toán"** with an equal split → shown as typed, not relabeled as a payment. (Task 3 `isPayment` test)
5. **Negative balances in English** → `-5,000₫` (sign before number). (Task 1 test)

Replace `lib/i18n/index.ts`:

```ts
import { en } from './en';
import { vi } from './vi';

export type Dict = typeof vi;
export type Locale = 'vi' | 'en';

export const LOCALES: readonly Locale[] = ['vi', 'en'];
export const DEFAULT_LOCALE: Locale = 'vi';
export const LOCALE_COOKIE = 'locale';

export const dicts: Record<Locale, Dict> = { vi, en };

export const isLocale = (v: unknown): v is Locale => LOCALES.includes(v as Locale);

// cookie (explicit choice) → Accept-Language (highest q first) → vi
export function pickLocale(cookie: string | undefined, acceptLanguage: string | null): Locale {
  if (isLocale(cookie)) return cookie;
  const langs = (acceptLanguage ?? '')
    .split(',')
    .map((part) => {
      const [tag, ...attrs] = part.trim().split(';');
      const q = attrs.map((a) => a.trim()).find((a) => a.startsWith('q='));
      return { lang: tag.split('-')[0].toLowerCase(), q: q ? Number(q.slice(2)) : 1 };
    })
    .filter((l) => l.q > 0) // drops q=0 and unparsable q (NaN)
    .sort((a, b) => b.q - a.q); // stable: equal q keeps header order
  const match = langs.find((l) => isLocale(l.lang));
  return match ? (match.lang as Locale) : DEFAULT_LOCALE;
}
```

```ts
    .min(2, k('minMembers'))

  .object({

    amount: z.number().max(MAX_AMOUNT, k('amountTooLarge')).int().positive(k('amountPositive')),
    payerId: z.number().int(),
    splitType: z.enum(['equal', 'exact', 'shares']),
    splits: z
      .array(z.object({ memberId: z.number().int(), value: z.number().max(MAX_AMOUNT, k('splitValueTooLarge')).int().positive() }))
      .min(1, k('participantsRequired')),
  })
  .superRefine((e, ctx) => {
    const ids = e.splits.map((s) => s.memberId);
    if (new Set(ids).size !== ids.length) ctx.addIssue({ code: 'custom', message: k('duplicateParticipant') });
    if (e.splitType === 'shares' && e.splits.some((s) => s.value > MAX_SHARES)) {
      ctx.addIssue({ code: 'custom', message: k('maxShares') });
    }
    if (e.splitType === 'exact') {
      const total = e.splits.reduce((a, s) => a + s.value, 0);
      if (total !== e.amount) {
        ctx.addIssue({ code: 'custom', message: k('exactMismatch'), params: { total, amount: e.amount } });
      }
    }
  });

// zod built-in messages (Infinity, wrong type; only reachable with tampered input) are not keys → generic message
export function firstError(e: z.ZodError, t: Dict): string {
  const issue = e.issues[0];
  if (!issue || !Object.hasOwn(t.errors, issue.message)) return t.errors.invalid;
  const msg = t.errors[issue.message as ErrorKey];
  if (typeof msg === 'string') return msg;
```

```tsx
      {transfers.length === 0 && <p className="text-gray-500">{t.balances.settled}</p>}
      {transfers.map((tr) => (
        <div key={`${tr.from}-${tr.to}`} className="card flex items-center gap-3">
          <div className="flex-1">
            <b>{name.get(tr.from)}</b> → <b>{name.get(tr.to)}</b>
            <div className="font-semibold">{formatVnd(tr.amount, locale)}</div>
          </div>
          <form action={recordPayment.bind(null, id, tr.from, tr.to, tr.amount)}>
            <SubmitButton>{t.balances.paid}</SubmitButton>
          </form>
        </div>
      ))}
    </div>
  );
}
```

Note: the transfer loop variable is renamed `t` → `tr` so it does not shadow the dictionary.
